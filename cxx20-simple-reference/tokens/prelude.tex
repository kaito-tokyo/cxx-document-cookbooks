% Read before std.tex: run the draft with LuaLaTeX and write a token stream.
\RequirePackage{luatex85}
\chardef\pdfshellescape=0
\directlua{cxxtokens_dir = status.filename:match("^(.*)/[^/]*$") or "."}
\directlua{dofile(cxxtokens_dir .. "/dummy-fonts.lua")}
\directlua{dofile(cxxtokens_dir .. "/cxxtokens.lua")}
\AddToHook{package/graphicx/after}{\RenewDocumentCommand\includegraphics{s O{} m}{[#3]}}

% Older drafts (N4861) replace an internal of listings in styles.tex and
% stop unless listings is 1.5b to 1.8c; pretend 1.8c while styles.tex is read.
\AddToHook{file/styles.tex/before}{\expandafter\let\csname cxxlstversion\expandafter\endcsname
  \csname lst@version\endcsname \expandafter\def\csname lst@version\endcsname{1.8c}}
\AddToHook{file/styles.tex/after}{\expandafter\let\csname lst@version\expandafter\endcsname
  \csname cxxlstversion\endcsname}

% Older drafts (N4861) \renewcommand memoir's \tref and \fref, which recent
% memoir no longer defines; recent drafts \providecommand them first.
\AddToHook{file/macros.tex/before}{\providecommand\tref{}\providecommand\fref{}}

% Older drafts (N4861) extract the grammar with the extract package, which
% replaces LaTeX's \begin and \end with copies of those of LaTeX 2019 to run
% \AfterEndEnv hooks. That bypasses LaTeX's environment hooks (so no note,
% example or codeblock marks), and with recent LaTeX it loops for ever. Keep
% LaTeX's \begin and \end, and run \AfterEndEnv code through LaTeX's own
% env/NAME/after hook instead.
\makeatletter
\AddToHook{package/extract/before}{%
  \expandafter\let\csname cxx@begin\expandafter\endcsname\csname begin \endcsname
  \expandafter\let\csname cxx@end\expandafter\endcsname\csname end \endcsname}
\AddToHook{package/extract/after}{%
  \expandafter\let\csname begin \expandafter\endcsname\csname cxx@begin\endcsname
  \expandafter\let\csname end \expandafter\endcsname\csname cxx@end\endcsname
  \def\AfterEndEnv#1{\edef\cxx@hook{env/\@currenvir/after}%
    \expandafter\AddToHookNext\expandafter{\cxx@hook}{#1}}}
\makeatother

% Walk each page into the token stream, then empty it of all but its
% \write nodes and destinations.
\AddToHook{shipout/before}{\directlua{cxx.shipout(\number\ShipoutBox)}}
\AddToHook{enddocument/end}{\directlua{cxx.finish()}}

% A mark in the token stream. Protected, so that it is written as is to
% .aux/.toc instead of being executed there.
\newif\ifcxxmarks \cxxmarkstrue
\protected\def\cxxmark#1{\ifcxxmarks\directlua{cxx.mark("\luaescapestring{#1}")}\fi}

% Titles carry marks into .toc; the table of contents is typeset without them.
\AddToHook{cmd/tableofcontents/before}{\global\cxxmarksfalse}
\AddToHook{cmd/tableofcontents/after}{\global\cxxmarkstrue}

% No hyphenation, no running heads.
\AtBeginDocument{\hyphenpenalty=10000 \exhyphenpenalty=10000 \pagestyle{empty}%
  \RenewDocumentCommand\pagestyle{m}{}\RenewDocumentCommand\thispagestyle{m}{}}

% Sections: level, abbreviation, and the title between marks.
\AtBeginDocument{%
  \let\cxxorigrSec\rSec
  \def\rSec#1[#2]#3{\cxxmark{section #1 #2}\cxxorigrSec#1[#2]{\cxxmark{begin title}#3\cxxmark{end title}}}%
}

% amsmath typesets an equation number several times and keeps a copy
% without hyperref's destination; name the destination once more.
\AddToHook{env/equation/end}{\cxxmark{anchor \csname @currentHref\endcsname}}

% The \glossary entries of xrefdelta.tex are \write nodes alone on a page
% that is otherwise empty until the glossary exists, so the page is never
% shipped out and the entries are lost. An empty box gives the page content.
\AddToHook{file/xrefdelta.tex/after}{\par\null\par}

% Parts of the back matter: bibliography, cross-references, indexes.
\AddToHook{env/thebibliography/begin}{\cxxmark{part}}
\AddToHook{cmd/printglossary/before}{\cxxmark{part}}
\AddToHook{cmd/printindex/before}{\cxxmark{part}}

% Paragraph numbers.
\AddToHook{cmd/pnum/before}{\cxxmark{pnum}}

% Notes, examples, code blocks.
\AddToHook{env/note/begin}{\cxxmark{begin note}}
\AddToHook{env/note/end}{\cxxmark{end note}}
\AddToHook{env/example/begin}{\cxxmark{begin example}}
\AddToHook{env/example/end}{\cxxmark{end example}}
\AddToHook{env/codeblock/begin}{\cxxmark{begin codeblock}}
\AddToHook{env/codeblock/end}{\cxxmark{end codeblock}}

% Inline markup. The commands are wrapped in a group rather than given
% cmd hooks, so that they still work as a subscript without braces
% ($E_\tcode{x}$ in older drafts).
\AtBeginDocument{%
  \NewCommandCopy\cxxorigtcode\tcode
  \RenewDocumentCommand\tcode{m}{{\cxxmark{begin tcode}\cxxorigtcode{#1}\cxxmark{end tcode}}}%
  \NewCommandCopy\cxxoriggrammarterm\grammarterm
  \RenewDocumentCommand\grammarterm{m}{{\cxxmark{begin grammarterm}\cxxoriggrammarterm{#1}\cxxmark{end grammarterm}}}%
  \NewCommandCopy\cxxorigdefnx\defnx
  \RenewDocumentCommand\defnx{mm}{{\cxxmark{begin defn}\cxxorigdefnx{#1}{#2}\cxxmark{end defn}}}%
}
\AtBeginDocument{%
  \NewCommandCopy\cxxorigiref\iref
  \RenewDocumentCommand\iref{m}{\cxxmark{iref #1}\cxxorigiref{#1}}%
}
